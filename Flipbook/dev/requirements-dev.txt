# Packages needed to run Flipbook from source and to build Flipbook.exe (Python 3.13, Windows x64).

pillow==12.3.0            # only for make_icon.py (excluded from the exe)
pypdf==7.0.1              # bundled since 1.8.17: legal.py and mydict.py read the user's own dictionary PDFs
